Guide & Setup

1. Declarative HTML Setup

To use the web component directly in HTML, place original and modified source codes inside child <script> tags. 

<!-- Import module to register custom element -->
<script type="module" src="./web-component/composite-monaco-diff.js"></script>

<composite-monaco-diff theme="vs-dark" language="javascript" style="height: 400px;">
  <script type="text/original" lang="javascript" data-offset="4">
    const value = 1;
  </script>
  <script type="text/modified" lang="javascript" data-offset="4">
    const value = 2;
  </script>
</composite-monaco-diff>

2. React Setup

Import the React wrapper component. Ensure parent layout has proper sizing or pass custom inline styles. 

import MonacoDiff from "./web-component/react";

export default function App() {
  return (
    <MonacoDiff
      theme="vs-dark"
      language="javascript"
      original="const a = 1;"
      modified="const a = 2;"
      style={{ height: "400px" }}
    />
  );
}

3. Imperative Access via Refs in React

Access the underlying manager to control/read Monaco elements programmatically. 

import React, { useRef, useEffect } from "react";
import MonacoDiff from "./web-component/react";
import { MonacoDiffElement } from "./web-component/composite-monaco-diff";

export default function ManagerExample() {
  const ref = useRef<MonacoDiffElement>(null);

  useEffect(() => {
    if (ref.current) {
      ref.current.whenReady().then(() => {
        const mgr = ref.current.getManager();
        const editor = mgr.getEditor();
        console.log("Monaco instance:", editor);
      });
    }
  }, []);

  return <MonacoDiff ref={ref} original="left code" modified="right code" style={{ height: "400px" }} />;
}
